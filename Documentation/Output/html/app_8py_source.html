29class UserManager:
30 def __init__(self, user_file):
31 self.user_file = user_file
32 self.users = self.load_users()

34 # This method Load users from the JSON file or return an empty list if the file doesn't exist.
35 def load_users(self):
36 """Load users from the JSON file, or return empty list if file doesn't exist."""
37 if os.path.exists(self.user_file):
38 with open(self.user_file, 'r') as f:
39 return json.load(f)
40 return []
41
42 # This method writes the users list to the specified JSON file.
43 def save_users(self):
44 """Save the current users list to the JSON file."""
45 with open(self.user_file, 'w') as f:
46 json.dump(self.users, f)
47
48 # This method adds a new user to the users list and saves it to the JSON file.
49 def add_user(self, name, email, password, role='Owner'):
50 """Add a new user and save."""
51 self.users.append({
52 'name': name,
53 'email': email,
54 'password': generate_password_hash(password),
55 'role': role
56 })
57 self.save_users()
58
59 # This method finds a user by email in the users list, It returns the user dictionary if found, or None if not found.
60 def find_user(self, email):
61 """Find a user by email."""
62 return next((u for u in self.users if u['email'] == email), None)
63
64 # This method checks if an email already exists in the users list, It returns True if the email is found, or False otherwise.
65 def email_exists(self, email):
66 """Check if an email is already registered."""
67 return any(u['email'] == email for u in self.users)

78class AIReportGenerator:
79 # This method generates a summary AI report from the filtered DataFrame.
80 @staticmethod
81 def generate(filtered_df):
82 """Generate a summary AI report from the filtered DataFrame."""
83 if filtered_df.empty:
84 return "No data available for AI report."
85 most_profitable_vehicle = filtered_df.groupby('Vehicle ID')['Net Profit'].sum().idxmax()
86 top_routes = ", ".join(filtered_df['Route'].value_counts().head(2).index) if 'Route' in filtered_df.columns else "N/A"
87 avg_profit_per_trip = round(filtered_df['Net Profit'].sum() / len(filtered_df), 2)
88 rev = filtered_df['Freight Amount'].sum()
89 exp = filtered_df['Total Trip Expense'].sum()
90 profit = filtered_df['Net Profit'].sum()
91 kms = filtered_df['Actual Distance (KM)'].sum()
92 profit_pct = round((profit / rev * 100), 1) if rev else 0
93 per_km = round(profit / kms, 2) if kms else 0
94 return f"""
95AI Report Highlights:
96
97Total Trips: {len(filtered_df)}
98On-going Trips: {filtered_df[filtered_df['Trip Status'] == 'Pending Closure'].shape[0]}
99Completed Trips: {filtered_df[filtered_df['Trip Status'] == 'Completed'].shape[0]}
100Profit Percentage: {profit_pct}%
101
102Financials:
103- Revenue: Rs{round(rev / 1e6, 2)}M
104- Expense: Rs{round(exp / 1e6, 2)}M
105- Profit: Rs{round(profit / 1e6, 2)}M
106- KMs Travelled: {round(kms / 1e3, 1)}K
107- Cost per KM: Rs{per_km}
108
109AI Insights:
110- Top Vehicle: {most_profitable_vehicle}
